cmake_minimum_required(VERSION 3.24)
project(AgentNodify VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8 /W4 /EHsc)
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Test)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json
  URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3
  URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406)
FetchContent_MakeAvailable(json)
set(JSON_VALIDATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JSON_VALIDATOR_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json_validator
  URL https://codeload.github.com/pboettch/json-schema-validator/tar.gz/refs/tags/2.3.0
  URL_HASH SHA256=2c00b50023c7d557cdaa71c0777f5bcff996c4efd7a539e58beaa4219fa2a5e1)
FetchContent_MakeAvailable(json_validator)
set(QT_NODES_DEVELOPER_DEFAULTS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(qtnodes
  URL https://codeload.github.com/paceholder/nodeeditor/tar.gz/refs/tags/3.0.12
  URL_HASH SHA256=67268eccdd389732447f4dfcf04ba3345d91a8728d38582642eab84017208522)
FetchContent_MakeAvailable(qtnodes)
add_library(agentcore src/flow.cpp src/runner.cpp src/flow.h src/runner.h src/coordinator.cpp src/coordinator.h src/cli.cpp src/cli.h src/modelcatalog.cpp src/modelcatalog.h src/history.cpp src/history.h src/progress.cpp src/progress.h)
target_include_directories(agentcore PUBLIC src)
target_link_libraries(agentcore PUBLIC Qt6::Core nlohmann_json::nlohmann_json nlohmann_json_schema_validator)
add_library(agentui src/window.cpp src/window.h src/graph.cpp src/graph.h src/nodeform.cpp src/nodeform.h src/theme.cpp src/theme.h src/graphview.cpp src/graphview.h src/projectopener.cpp src/projectopener.h src/progressview.cpp src/progressview.h)
target_link_libraries(agentui PUBLIC agentcore Qt6::Widgets QtNodes::QtNodes)
add_executable(AgentNodify WIN32 src/main.cpp)
target_sources(AgentNodify PRIVATE src/instancelock.h)
target_link_libraries(AgentNodify PRIVATE agentui)
enable_testing()
add_executable(agentnodify_instance_tests tests/instance_test.cpp src/instancelock.h)
target_link_libraries(agentnodify_instance_tests PRIVATE agentcore Qt6::Test)
add_test(NAME instance COMMAND agentnodify_instance_tests -o -,txt)
set_tests_properties(instance PROPERTIES TIMEOUT 30)
add_executable(agentnodify_tests tests/core_test.cpp tests/core_test.h)
target_link_libraries(agentnodify_tests PRIVATE agentcore Qt6::Test)
target_compile_definitions(agentnodify_tests PRIVATE AGENTNODIFY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME core COMMAND agentnodify_tests -o -,txt)
add_executable(agentnodify_script_tests tests/script_test.cpp tests/script_test.h)
target_link_libraries(agentnodify_script_tests PRIVATE agentcore Qt6::Test)
target_compile_definitions(agentnodify_script_tests PRIVATE AGENTNODIFY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME script COMMAND agentnodify_script_tests -o -,txt)
add_executable(agentnodify_shell_tests tests/shell_test.cpp tests/shell_test.h)
target_link_libraries(agentnodify_shell_tests PRIVATE agentcore Qt6::Test)
target_compile_definitions(agentnodify_shell_tests PRIVATE AGENTNODIFY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME shell COMMAND agentnodify_shell_tests -o -,txt)
add_executable(agentnodify_gui_tests tests/gui_test.cpp tests/gui_test.h)
target_link_libraries(agentnodify_gui_tests PRIVATE agentui Qt6::Test)
target_compile_definitions(agentnodify_gui_tests PRIVATE AGENTNODIFY_TEST_OUTPUT_DIR="${CMAKE_BINARY_DIR}")
add_test(NAME gui COMMAND agentnodify_gui_tests -o -,txt)
add_executable(fake_catalog tests/fake_catalog.cpp)
target_link_libraries(fake_catalog PRIVATE Qt6::Core nlohmann_json::nlohmann_json)
add_executable(agentnodify_catalog_tests tests/catalog_test.cpp tests/catalog_test.h)
target_link_libraries(agentnodify_catalog_tests PRIVATE agentui Qt6::Test)
add_dependencies(agentnodify_catalog_tests fake_catalog)
add_test(NAME catalog COMMAND agentnodify_catalog_tests -o -,txt)
add_executable(agentnodify_progress_tests tests/progress_test.cpp tests/progress_test.h)
target_link_libraries(agentnodify_progress_tests PRIVATE agentui Qt6::Test)
target_compile_definitions(agentnodify_progress_tests PRIVATE AGENTNODIFY_TEST_OUTPUT_DIR="${CMAKE_BINARY_DIR}")
add_test(NAME progress COMMAND agentnodify_progress_tests -o -,txt)
add_executable(agentnodify_progress_probe tests/progress_probe.cpp)
target_link_libraries(agentnodify_progress_probe PRIVATE agentcore)
add_executable(fake_agent tests/fake_agent.cpp)
target_link_libraries(fake_agent PRIVATE Qt6::Core)
add_dependencies(agentnodify_tests fake_agent)
add_executable(agentnodify_context_tests tests/context_test.cpp)
target_link_libraries(agentnodify_context_tests PRIVATE agentcore Qt6::Test)
add_dependencies(agentnodify_context_tests fake_agent)
add_test(NAME context COMMAND agentnodify_context_tests -o -,txt)
set_tests_properties(context PROPERTIES TIMEOUT 90)
add_executable(agentnodify_coordinator_tests tests/coordinator_test.cpp)
target_link_libraries(agentnodify_coordinator_tests PRIVATE agentcore Qt6::Test)
add_dependencies(agentnodify_coordinator_tests fake_agent)
add_test(NAME coordinator COMMAND agentnodify_coordinator_tests -o -,txt)
set_tests_properties(coordinator PROPERTIES TIMEOUT 90)
add_dependencies(agentnodify_gui_tests fake_agent)
add_executable(agentnodify_live_test tests/live_test.cpp)
target_link_libraries(agentnodify_live_test PRIVATE agentcore)
add_executable(agentnodify_catalog_probe tests/catalog_probe.cpp)
target_link_libraries(agentnodify_catalog_probe PRIVATE agentcore)
add_test(NAME gui_smoke COMMAND AgentNodify --smoke-test)
get_target_property(_qmake Qt6::qmake IMPORTED_LOCATION)
execute_process(COMMAND "${_qmake}" -query QT_INSTALL_PLUGINS
  OUTPUT_VARIABLE _qt_plugins OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(_gui_test_env "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${_qt_plugins}")
if(WIN32)
  file(TO_CMAKE_PATH "$ENV{WINDIR}/Fonts" _windows_fonts)
  list(APPEND _gui_test_env "QT_QPA_FONTDIR=${_windows_fonts}")
endif()
set_tests_properties(gui_smoke gui catalog progress PROPERTIES
  ENVIRONMENT "${_gui_test_env}"
  TIMEOUT 20)
set_tests_properties(core PROPERTIES TIMEOUT 90)
set_tests_properties(script PROPERTIES TIMEOUT 90)
set_tests_properties(shell PROPERTIES TIMEOUT 120)
install(TARGETS AgentNodify RUNTIME DESTINATION bin)
